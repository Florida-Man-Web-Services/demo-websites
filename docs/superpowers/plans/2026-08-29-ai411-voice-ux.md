import importlib
import os
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))


def _reload():
    os.environ["AGENT_MODE"] = "ai411"
    import config, ai411
    importlib.reload(config)
    importlib.reload(ai411)
    return ai411


def test_greeting_unchanged():
    ai411 = _reload()
    assert ai411.AI411_GREETING == "A411 here."


def test_openers_have_no_filler():

    banned = ("good question", "great question", "sure thing", "absolutely",

    for b in banned:
